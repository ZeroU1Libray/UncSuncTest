local env = {
	math = math,
	pi = math.pi,
	e = math.exp(1),
	sin = math.sin,
	cos = math.cos,
	tan = math.tan,
	asin = math.asin,
	acos = math.acos,
	atan = math.atan,
	sqrt = math.sqrt,
	log = function(x) return math.log(x, 10) end,
	ln = math.log,
	abs = math.abs,
	exp = math.exp,
	floor = math.floor,
	ceil = math.ceil,
	fact = function(n)
		if n < 0 then return nil end
		local r = 1
		for i = 2, math.floor(n) do 
			r = r * i 
		end
		return r
	end
}

local function convertFactorialSyntax(expr)
	local result = expr:gsub("(%d*%.?%d+)!", "env.fact(%1)")
	return result
end

local function eval(expr)
	if expr == "" then return "" end
	expr = expr:gsub("%s+", "")
	expr = convertFactorialSyntax(expr)
	local function checkFunctionCalls(str)
		local functions = {"sin", "cos", "tan", "asin", "acos", "atan", "sqrt", "log", "ln", "abs", "exp", "floor", "ceil"}
		for _, func in ipairs(functions) do
			if str:sub(-#func) == func and not str:find(func.."%(") then
				str = str .. "()"
			end
		end
		return str
	end
	expr = checkFunctionCalls(expr)
	local func, err = loadstring("return " .. expr)
	if not func then
		func, err = loadstring("local result = " .. expr .. "; return result")
		if not func then
			return "Syntax Error: " .. err
		end
	end
	setfenv(func, env)
	local success, result = pcall(func)
	if success then
		local currentExpression = Display.Text
		if History.Text == "History:" then
			History.Text = History.Text .. "\n" .. currentExpression .. " = " .. tostring(result)
		else
			History.Text = History.Text .. "\n" .. currentExpression .. " = " .. tostring(result)
		end
		return tostring(result)
	else
		return "Math Error"
	end
end

local function createButton(text, x, y)
	local btn = Instance.new("TextButton", Main)
	btn.Size = UDim2.fromOffset(62, 40)
	btn.Position = UDim2.fromOffset(10 + x * 68, 180 + y * 45)
	btn.Text = text
	btn.Font = Enum.Font.SourceSansBold
	btn.TextSize = 18
	btn.TextColor3 = Color3.new(1, 1, 1)
	btn.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
	Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 6)

	btn.MouseButton1Click:Connect(function()
		if text == "=" then
			local result = eval(Display.Text)
			Display.Text = result
		elseif text == "C" then
			Display.Text = ""
			History.Text = "History:"
		elseif text == "^" then
			Display.Text = Display.Text .. "^"
		elseif text == "sqrt" or text == "sin" or text == "cos" or text == "tan" or 
		       text == "log" or text == "ln" or text == "abs" or text == "exp" or 
		       text == "floor" or text == "ceil" then
			Display.Text = Display.Text .. text .. "("
		elseif text == "pi" then
			Display.Text = Display.Text .. "pi"
		else
			Display.Text = Display.Text .. text
		end
	end)
end

local x, y = 0, 0
for _, v in ipairs(buttons) do
	createButton(v, x, y)
	x = x + 1
	if x >= 5 then
		x = 0
		y = y + 1
	end
end
